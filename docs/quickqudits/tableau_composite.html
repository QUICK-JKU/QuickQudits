quickqudits.tableau_composite

Composite-dimension tableau measurement via diagonal-subgroup extraction and SNF.

This module provides the CompositeTableau class, which inherits from tableau.Tableau but overrides measurement for arbitrary dimensions d.

Computational-basis measurement outcomes x ∈ Z_d^n are characterized by the diagonal subgroup of the stabilizer group. A product of generators specified by coefficients u ∈ Z_d^n is diagonal iff its X-part cancels: u^T X_s ≡ 0 (mod d), where X_s are the stabilizer-generator X rows.

For each diagonal stabilizer with Z-exponents b and even phase exponent t, valid outcomes satisfy the linear congruence b · x ≡ -t/2 (mod d).

Stacking all such constraints yields B x ≡ c (mod d), which is solved and sampled exactly using Smith normal decomposition (SNF) provided by SymPy: D = U B V, reducing the system to independent 1D congruences in the transformed variables.

 38def _np_to_sympy(A: np.ndarray) -> Matrix:
 39    """
 40    Convert a NumPy integer array to a SymPy Matrix (exact integers).
 41
 42    Parameters
 43    ----------
 44    A : numpy.ndarray
 45        Integer array.
 46
 47    Returns
 48    -------
 49    sympy.Matrix
 50        Matrix with Python ``int`` entries.
 51    """
 52    return Matrix([[int(v) for v in row] for row in A.tolist()])

 55def _sympy_to_np(A: Matrix, mod=None) -> np.ndarray:
 56    """
 57    Convert a SymPy Matrix to a NumPy integer array.
 58
 59    Parameters
 60    ----------
 61    A : sympy.Matrix
 62        Integer matrix.
 63
 64    Returns
 65    -------
 66    numpy.ndarray

 76def _snf_int(
 77    A: np.ndarray, mod: Optional[int] = None
 78) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
 79    """
 80    Smith normal decomposition over the integers using SymPy.
 81
 82    Parameters
 83    ----------
 84    A : numpy.ndarray
 85        Integer matrix.
 86
 87    Returns
 88    -------
 89    D : numpy.ndarray
 90        Diagonal Smith form.
 91    U : numpy.ndarray
 92        Left unimodular transform.
 93    V : numpy.ndarray
 94        Right unimodular transform.
 95
 96    Notes
 97    -----
 98    Uses SymPy's ``smith_normal_decomp`` which returns matrices satisfying
 99        D = U * A * V .
100    """
101    M = _np_to_sympy(A)
102    D, U, V = smith_normal_decomp(M)
103    if mod is None:
104        return _sympy_to_np(D), _sympy_to_np(U), _sympy_to_np(V)
105
106    return _sympy_to_np(D, mod=mod), _sympy_to_np(U, mod=mod), _sympy_to_np(V, mod=mod)

109def _kernel_generators_mod_d_snf(A: np.ndarray, d: int) -> np.ndarray:
110    """
111    Compute a generating set for the solution module of:
112
113        A x ≡ 0 (mod d)
114
115    using integer SNF.
116
117    Parameters
118    ----------
119    A : numpy.ndarray
120        Integer matrix of shape (m, n).
121    d : int
122        Modulus.
123
124    Returns
125    -------
126    numpy.ndarray
127        Array of shape (k, n) whose rows generate the kernel over Z_d.
128    """
129    A = np.array(A, dtype=int)
130    m, n = A.shape
131
132    if n == 0:
133        return np.zeros((0, 0), dtype=int)
134
135    D, U, V = _snf_int(A, mod=d)
136
137    gens_y: List[np.ndarray] = []
138
139    r = min(m, n)
140    for i in range(r):
141        s = abs(int(D[i, i]))
142        if s == 0:
143            e = np.zeros(n, dtype=int)
144            e[i] = 1
145            gens_y.append(e)
146        else:
147            g = int(np.gcd(s, d))
148            step = d // g
149            e = np.zeros(n, dtype=int)
150            e[i] = step % d
151            gens_y.append(e)
152
153    for i in range(r, n):
154        e = np.zeros(n, dtype=int)
155        e[i] = 1
156        gens_y.append(e)
157
158    if len(gens_y) == 0:
159        return np.zeros((0, n), dtype=int)
160
161    GY = np.vstack(gens_y) % d
162    X = (V @ GY.T) % d
163    return X.T.astype(int, copy=False)

166def _solve_congruences_mod_d_snf(
167    B: np.ndarray, c: np.ndarray, d: int
168) -> Tuple[np.ndarray, np.ndarray, List[Tuple[int, int, int]], List[int]]:
169    """
170    Solve B x ≡ c (mod d) using integer SNF.
171
172    This returns a sampler description for the solution set, sufficient to
173    sample uniformly without enumerating all solutions.
174
175    Parameters
176    ----------
177    B : numpy.ndarray
178        Constraint matrix of shape (m, n).
179    c : numpy.ndarray
180        RHS vector of shape (m,).
181    d : int
182        Modulus.
183
184    Returns
185    -------
186    V : numpy.ndarray
187        Unimodular matrix from SNF, used to map y -> x via x = V y (mod d).
188    y0 : numpy.ndarray
189        One particular solution y0 in Z_d^n for the transformed system D y ≡ c' (mod d).
190    diag_info : list[tuple[int, int, int]]
191        For each i < r with diagonal s_i != 0, stores (i, modulus_red, g),
192        where:
193            - modulus_red = d / gcd(s_i, d)
194            - g = gcd(s_i, d)
195        Sampling rule: y_i = y0_i + modulus_red * k, k uniform in {0,...,g-1}
196    free_indices : list[int]
197        Indices of y that are free uniform in {0,...,d-1}.
198    """
199    B = np.array(B, dtype=int)
200    c = np.array(c, dtype=int).reshape(-1)
201
202    m, n = B.shape
203
204    D, U, V = _snf_int(B, mod=d)
205
206    cp = (U @ c.reshape(-1, 1)).reshape(-1)
207    cp = np.mod(cp, d).astype(int)
208
209    y0 = np.zeros(n, dtype=int)
210    diag_info: List[Tuple[int, int, int]] = []
211    free_indices: List[int] = []
212
213    r = min(m, n)
214
215    for i in range(r):
216        s = int(D[i, i])
217        rhs = int(cp[i] % d)
218
219        if s == 0:
220            free_indices.append(i)
221            continue
222
223        s_abs = abs(s)
224        g = int(np.gcd(s_abs, d))
225
226        s_red = (s // g) % (d // g)
227        rhs_red = (rhs // g) % (d // g)
228        mod_red = d // g
229
230        inv = pow(int(s_red), -1, int(mod_red))
231        yi0 = (inv * rhs_red) % mod_red
232
233        y0[i] = int(yi0)
234        diag_info.append((i, mod_red, g))
235
236    for i in range(r, n):
237        free_indices.append(i)
238
239    return V % d, y0 % d, diag_info, free_indices

242class CompositeTableau(Tableau):
243    """
244    Composite-dimension stabilizer tableau with exact measurement via SNF.
245
246    Notes
247    -----
248    This class implements computational-basis measurement for arbitrary dimensions ``d`` by:
249
250    1. Extracting the diagonal subgroup of the stabilizer group:
251       find exponent vectors ``u`` such that combining stabilizer generators with
252       exponents ``u`` cancels the X-part, producing a diagonal stabilizer.
253
254    2. Converting diagonal stabilizers into linear congruence constraints on outcomes:
255       for diagonal element with Z-exponents ``b`` and phase ``tau_exp`` (even),
256       valid outcomes ``x`` satisfy ``b·x ≡ -tau_exp/2 (mod d)``.
257
258    3. Solving and sampling from the congruence system using Smith normal form (SNF).
259    """
260
261    def _diagonal_constraints(self) -> Optional[Tuple[np.ndarray, np.ndarray]]:
262        """
263        Construct diagonal stabilizer constraints B x ≡ c (mod d).
264
265        Returns
266        -------
267        (B, c) or None
268            If diagonal constraints exist, returns:
269            - B : ndarray, shape (m, n)
270            - c : ndarray, shape (m,)
271            such that valid outcomes x satisfy B x ≡ c (mod d).
272            If no diagonal stabilizers exist, returns None (uniform distribution).
273        """
274        if not self.full:
275            raise ValueError("Composite measurement requires full tableau.")
276
277        n = self.n
278        d = self.d
279
280        Xs = (self.X[n:, :] % d).astype(int)
281
282        U = _kernel_generators_mod_d_snf(Xs.T, d)
283        if U.shape[0] == 0:
284            return None
285
286        diag_Z: List[np.ndarray] = []
287        diag_tau: List[int] = []
288
289        for u in U:
290            acc_X = np.zeros(n, dtype=self._xz_dtype)
291            acc_Z = np.zeros(n, dtype=self._xz_dtype)
292            acc_tau = 0
293
294            for j in range(n):
295                mj = int(u[j] % d)
296                if mj == 0:
297                    continue
298
299                Xp, Zp, taup = self._power(
300                    self.X[n + j], self.Z[n + j], int(self.tau_exp[n + j]), mj
301                )
302                acc_X, acc_Z, acc_tau = self._product_logic(
303                    acc_X, acc_Z, acc_tau, Xp, Zp, taup
304                )
305
306            if np.any((acc_X % d) != 0):
307                continue
308
309            t = int(acc_tau % (2 * d))
310
311            diag_Z.append((acc_Z % d).astype(int))
312            diag_tau.append(t)
313
314        if len(diag_Z) == 0:
315            return None
316
317        B = np.vstack(diag_Z) % d
318        c = (-np.array(diag_tau, dtype=int) // 2) % d
319        return B, c

321    def sample_measurements(
322        self,
323        shots: int,
324        rng: Optional[np.random.Generator] = None,
325        seed: Optional[int] = None,
326    ) -> np.ndarray:
327        """
328        Sample computational-basis measurement outcomes for all qudits.
329
330        The sampling is non-destructive: it does not update the tableau to a collapsed state.
331        Outcomes are sampled exactly from the stabilizer state's computational-basis
332        distribution for arbitrary dimensions ``d``.
333
334        Parameters
335        ----------
336        shots : int
337            Number of samples to draw.
338        rng : numpy.random.Generator, optional
339            Random number generator to use. If ``None``, a new generator is created.
340        seed : int, optional
341            Seed used to initialize the generator when ``rng`` is ``None``.
342
343        Returns
344        -------
345        numpy.ndarray
346            Integer array of shape ``(shots, n)`` where each row is an outcome
347            in ``{0, ..., d-1}^n``.
348        """
349        if shots < 1:
350            raise ValueError(f"shots must be >= 1, got {shots}")
351
352        rng = np.random.default_rng(seed) if rng is None else rng
353
354        n = self.n
355        d = self.d
356
357        cons = self._diagonal_constraints()
358        if cons is None:
359            return rng.integers(0, d, size=(shots, n), endpoint=False)
360
361        B, c = cons
362
363        V, y0, diag_info, free_indices = _solve_congruences_mod_d_snf(B, c, d)
364
365        samples = np.empty((shots, n), dtype=int)
366
367        for s in range(shots):
368            y = np.zeros(n, dtype=int)
369
370            y[:] = y0
371
372            for i, mod_red, g in diag_info:
373                if g == 1:
374                    continue
375                k = int(rng.integers(0, g))
376                y[i] = (y0[i] + (mod_red % d) * k) % d
377
378            for i in free_indices:
379                y[i] = int(rng.integers(0, d))
380
381            x = (V @ y) % d
382            samples[s] = x.astype(int, copy=False)
383
384        return samples

386    def sample_measurements_noisy_pf(
387        self,
388        qc,
389        shots: int,
390        rng: Optional[np.random.Generator] = None,
391        seed: Optional[int] = None,
392    ) -> np.ndarray:
393        """
394        This method overrides the corresponding method of tableau.Tableau,
395        so that it works for composite d.
396
397        Parameters
398        ----------
399        qc : QuantumCircuit
400            Circuit (with NoiseChannels).
401        shots : int
402            Number of samples to use for the estimate.
403        rng : np.random.Generator, optional
404            RNG to use. If None, one is created from `seed`.
405        seed : int, optional
406            Seed to create a fresh RNG if `rng` is None.
407
408        Returns
409        -------
410        np.ndarray
411            An integer array of shape (shots, qc.n), where each row is one sampled outcome
412            vector in {0, 1, ..., d-1}^n (mod d).
413
414        Raises
415        ------
416        ValueError
417            If `shots <= 0`.
418        """
419        self._check_compat_with_circuit(qc)
420
421        if shots <= 0:
422            raise ValueError("shots must be a positive integer.")
423
424        rng = self._coerce_rng(rng, seed)
425        n, d = qc.n, qc.d
426
427        T_clean, _ = type(self).clean_run(qc)
428
429        ref_rng = np.random.default_rng(rng.integers(0, 2**63 - 1, dtype=np.int64))
430        r = T_clean.measure_all(rng=ref_rng)
431        r = np.asarray(r, dtype=np.int64) % d
432
433        x, z = type(self)._pf_init_frames(shots, n, d, rng)
434
435        for gate, qudits, dagger in qc._ops:
436            if isinstance(gate, NoiseChannel):
437                q = int(qudits[0])
438                probs = qc.noise_registry.get(gate.model_id)
439                type(self)._pf_apply_noise_channel(x, z, q=q, probs=probs, d=d, rng=rng)
440                continue
441
442            key = type(self)._pf_gate_key(gate)
443            type(self)._pf_apply_gate(x, z, key=key, qudits=qudits, dagger=dagger, d=d)
444
445        return (r[None, :] + x) % d

Composite-dimension stabilizer tableau with exact measurement via SNF.

Notes

This class implements computational-basis measurement for arbitrary dimensions d by:

  1. Extracting the diagonal subgroup of the stabilizer group: find exponent vectors u such that combining stabilizer generators with exponents u cancels the X-part, producing a diagonal stabilizer.

  2. Converting diagonal stabilizers into linear congruence constraints on outcomes: for diagonal element with Z-exponents b and phase tau_exp (even), valid outcomes x satisfy b·x ≡ -tau_exp/2 (mod d).

  3. Solving and sampling from the congruence system using Smith normal form (SNF).

323        self,

326        seed: Optional[int] = None,
327    ) -> np.ndarray:
328        """
329        Sample computational-basis measurement outcomes for all qudits.
330
331        The sampling is non-destructive: it does not update the tableau to a collapsed state.
332        Outcomes are sampled exactly from the stabilizer state's computational-basis
333        distribution for arbitrary dimensions ``d``.
334
335        Parameters
336        ----------
337        shots : int
338            Number of samples to draw.
339        rng : numpy.random.Generator, optional
340            Random number generator to use. If ``None``, a new generator is created.
341        seed : int, optional
342            Seed used to initialize the generator when ``rng`` is ``None``.
343
344        Returns
345        -------
346        numpy.ndarray
347            Integer array of shape ``(shots, n)`` where each row is an outcome
348            in ``{0, ..., d-1}^n``.
349        """
350        if shots < 1:
351            raise ValueError(f"shots must be >= 1, got {shots}")
352
353        rng = np.random.default_rng(seed) if rng is None else rng
354
355        n = self.n
356        d = self.d
357
358        cons = self._diagonal_constraints()
359        if cons is None:
360            return rng.integers(0, d, size=(shots, n), endpoint=False)
361
362        B, c = cons
363
364        V, y0, diag_info, free_indices = _solve_congruences_mod_d_snf(B, c, d)
365
366        samples = np.empty((shots, n), dtype=int)
367
368        for s in range(shots):
369            y = np.zeros(n, dtype=int)
370
371            y[:] = y0
372
373            for i, mod_red, g in diag_info:
374                if g == 1:
375                    continue
376                k = int(rng.integers(0, g))
377                y[i] = (y0[i] + (mod_red % d) * k) % d
378
379            for i in free_indices:
380                y[i] = int(rng.integers(0, d))
381
382            x = (V @ y) % d
383            samples[s] = x.astype(int, copy=False)
384
385        return samples
