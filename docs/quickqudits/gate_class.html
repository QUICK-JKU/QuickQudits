 26class Gate:
 27    """
 28    Represents a quantum gate or channel.
 29
 30    A Gate object stores the metadata (name, dimension) and the logic
 31    (matrix functions) required to simulate the operation.
 32
 33    Parameters
 34    ----------
 35    name : str
 36        The name of the gate (e.g., "X", "CNOT").
 37    n_qudits : int
 38        The number of qudits this gate acts on.
 39    matrix_fn : Callable[[int], np.ndarray] or None
 40        A function that takes dimension `d` and returns the unitary matrix.
 41        Can be None for placeholder gates (like Measure).
 42    dagger_matrix_fn : Callable[[int], np.ndarray] or None, optional
 43        A function that returns the conjugate transpose matrix.
 44    apply_fn : Callable or None, optional
 45        An optional optimized function for applying the gate directly to a
 46        statevector without constructing the full matrix.
 47    color : str or None, optional
 48        Hex color code for visualization.
 49    a : float or None, optional
 50        Parameter 'a' for Weyl gates.
 51    b : float or None, optional
 52        Parameter 'b' for Weyl gates.
 53    """
 54
 55    def __init__(
 56        self,
 57        name: str,
 58        n_qudits: int,
 59        matrix_fn: Optional[Callable[[int], np.ndarray]],
 60        dagger_matrix_fn: Optional[Callable[[int], np.ndarray]] = None,
 61        apply_fn: Optional[Callable] = None,
 62        color: Optional[str] = None,
 63        a: Optional[float] = None,
 64        b: Optional[float] = None,
 65    ):
 66        self.name = name
 67        self.n_qudits = n_qudits
 68        self.matrix_fn = matrix_fn
 69        self.dagger_matrix_fn = dagger_matrix_fn
 70        self._apply_fn = apply_fn
 71        self.color = color
 72        self.a = a
 73        self.b = b

 76        return f"Gate(name={self.name!r}, n_qudits={self.n_qudits})"

115    def matrix(self, d: int, dagger: bool = False) -> np.ndarray:
116        """
117        Return the numerical matrix for this gate at dimension d.
118
119        Parameters
120        ----------
121        d : int
122            Dimension of the qudit.
123        dagger : bool, default=False
124            If True, return the conjugate transpose matrix.
125
126        Returns
127        -------
128        np.ndarray
129            The unitary matrix.
130
131        Raises
132        ------
133        ValueError
134            If the gate does not define a matrix function.
135        """
136        if dagger:
137            if not self.dagger_matrix_fn:
138                raise ValueError(f"No dagger matrix available for gate {self.name}")
139            return self.dagger_matrix_fn(d)
140        if self.matrix_fn is None:
141            raise ValueError(f"No matrix available for gate {self.name}")
142        return self.matrix_fn(d)

144    def apply(
145        self,
146        psi: np.ndarray,
147        n: int,
148        d: int,
149        qudits: tuple[int, ...],
150        dagger: bool = False,
151    ) -> np.ndarray:
152        """
153        Apply this gate to a statevector.
154
155        Uses the optimized `apply_fn` if available; otherwise falls back to
156        tensor contraction using the matrix definition.
157
158        Parameters
159        ----------
160        psi : np.ndarray
161            Statevector of length d^n.
162        n : int
163            Total number of qudits.
164        d : int
165            Dimension of qudits.
166        qudits : tuple[int, ...]
167            Indices of the target qudits.
168        dagger : bool, default=False
169            If True, apply the inverse.
170
171        Returns
172        -------
173        np.ndarray
174            The updated statevector.
175        """
176        # Use optimized apply function if available
177        if self._apply_fn is not None:
178            if self.a is not None and self.b is not None:
179                # Weyl gate special case
180                return self._apply_fn(psi, n, d, qudits, dagger, a=self.a, b=self.b)
181            else:
182                # CNOT/CZ/SWAP
183                return self._apply_fn(psi, n, d, qudits, dagger)
184
185        # Optimized 1-qudit system case
186        if n == 1:
187            U = self.matrix(d, dagger)
188            return U @ psi
189
190        # General Tensor Contraction
191        Umat = self.dagger_matrix_fn(d) if dagger else self.matrix_fn(d)
192        k = self.n_qudits
193
194        U = Umat.reshape((d,) * k + (d,) * k)
195        psi_nd = psi.reshape((d,) * n)
196        axes_U = list(range(k, 2 * k))
197        axes_psi = list(qudits)
198        contracted = np.tensordot(U, psi_nd, axes=(axes_U, axes_psi))
199        rest = [i for i in range(n) if i not in qudits]
200
201        perm = []
202        for i in range(n):
203            if i in qudits:
204                perm.append(qudits.index(i))
205            else:
206                perm.append(k + rest.index(i))
207
208        out = np.transpose(contracted, perm)
209        return out.reshape(-1)

215    @staticmethod
216    def _apply_operator_to_density_matrix(
217        rho: np.ndarray, operator: np.ndarray, qudits: tuple[int, ...], n: int, d: int
218    ) -> np.ndarray:
219        """
220        Applies rho' = U * rho * U_dagger by explicitly constructing the full d^n x d^n operator.
221
222        Parameters
223        ----------
224        rho : np.ndarray
225            Density matrix of shape (d^n, d^n).
226        operator : np.ndarray
227            The d^k x d^k matrix of the small gate.
228        qudits : tuple[int, ...]
229            Target qudits.
230        n : int
231            Total number of qudits.
232        d : int
233            Dimension of qudits.
234
235        Returns
236        -------
237        np.ndarray
238            The updated density matrix.
239        """
240        from .circuit import QuantumCircuit
241
242        k = len(qudits)
243
244        # Create a temporary gate object that holds the small operator.
245        temp_gate = Gate("temp", k, lambda d_val: operator)
246
247        # Create a temporary circuit to access the _get_gate_unitary helper.
248        temp_qc = QuantumCircuit(n, d)
249
250        # Use the helper to construct the full d^n x d^n operator.
251        U = temp_qc._get_gate_unitary(temp_gate, qudits, False)
252
253        # Get the conjugate transpose for the right-multiplication.
254        U_dag = np.conjugate(U.T)
255
256        # Perform the matrix multiplication.
257        return U @ rho @ U_dag
258
259    def apply_density(
260        self,
261        rho: np.ndarray,
262        circuit: QuantumCircuit,
263        qudits: tuple[int, ...],
264        dagger: bool = False,
265    ) -> np.ndarray:
266        """
267        Applies a gate to a density matrix: rho' = U rho U†.
268
269        Parameters
270        ----------
271        rho : np.ndarray
272            Density matrix of shape (d^n, d^n).
273        circuit : QuantumCircuit
274            Reference to the circuit (for accessing n, d).
275        qudits : tuple[int, ...]
276            Target qudits.
277        dagger : bool, default=False
278            If True, apply inverse.
279
280        Returns
281        -------
282        np.ndarray
283            Updated density matrix.
284        """
285        if self.matrix_fn is None:
286            return rho
287
288        U = self.matrix(circuit.d, dagger)
289
290        return Gate._apply_operator_to_density_matrix(
291            rho, U, qudits, circuit.n, circuit.d
292        )

300class NoiseChannel(Gate):
301    """
302    Represents a noise channel.
303
304    Unlike standard gates, this does not have a static matrix. Instead, it
305    stores a `model_id` which points to a noise model in the circuit's
306    registry.
307
308    Parameters
309    ----------
310    model_id : int
311        The ID of the noise model in the circuit registry.
312    color : str or None, optional
313        Visualization color. Opacity corresponds to error probability.
314    """
315
316    def __init__(self, model_id: int, color: Optional[tuple] = None):
317        # Call the parent Gate constructor with fixed properties for a noise channel
318        super().__init__(name="NOISE_CHANNEL", n_qudits=1, matrix_fn=None, color=color)
319        self.model_id = model_id
320
321    def apply_density(
322        self,
323        rho: np.ndarray,
324        circuit: QuantumCircuit,
325        qudits: tuple[int, ...],
326        dagger: bool = False,
327    ) -> np.ndarray:
328        """
329        Apply the noise channel to a density matrix using using the
330        Kraus operator sum representation. Implements on-the-fly computation
331        and caching of Kraus operators.
332
333        rho' = sum(E_k * rho * E_k†)
334
335        Parameters
336        ----------
337        rho : np.ndarray
338            Density matrix of shape (d^n, d^n).
339        circuit : QuantumCircuit
340            Reference to the circuit.
341        qudits : tuple[int, ...]
342            Target qudits.
343        dagger : bool
344            Ignored for noise channels.
345
346        Returns
347        -------
348        np.ndarray
349            Updated density matrix.
350        """
351        # Check the circuit's cache for our Kraus operators.
352        kraus_operators = circuit._kraus_operator_cache.get(self.model_id)
353
354        # If not in the cache, compute and store them.
355        if kraus_operators is None:
356            # Retrieve the probability array from the permanent registry.
357            prob_array = circuit.noise_registry.get(self.model_id)
358            if prob_array is None:
359                # This should not happen if the circuit is constructed correctly.
360                return rho
361
362            d = circuit.d
363            prob_matrix = prob_array.reshape((d, d))
364
365            # Compute the list of d x d Kraus operators.
366            kraus_operators = []
367            for a in range(d):
368                for b in range(d):
369                    prob = prob_matrix[a, b]
370                    if (
371                        prob > 1e-12
372                    ):  # Only create operators for non-zero probabilities.
373                        # E_k = sqrt(p_k) * W_k
374                        kraus_op = np.sqrt(prob) * gates.W(d, a, b)
375                        kraus_operators.append(kraus_op)
376
377            # Store the computed list in the cache for future use in this run.
378            circuit._kraus_operator_cache[self.model_id] = kraus_operators
379
380        # Apply the Kraus sum: rho' = sum(E_k * rho * E_k_dagger)
381        rho_new = np.zeros_like(rho)
382        for E_k in kraus_operators:
383            term = Gate._apply_operator_to_density_matrix(
384                rho, E_k, qudits, circuit.n, circuit.d
385            )
386            rho_new += term
387
388        return rho_new

164            Total number of qudits.
165        d : int
166            Dimension of qudits.
167        qudits : tuple[int, ...]
168            Indices of the target qudits.
169        dagger : bool, default=False
170            If True, apply the inverse.
171
172        Returns
173        -------
174        np.ndarray
175            The updated statevector.
176        """
177        # Use optimized apply function if available
178        if self._apply_fn is not None:
179            if self.a is not None and self.b is not None:
180                # Weyl gate special case
181                return self._apply_fn(psi, n, d, qudits, dagger, a=self.a, b=self.b)
182            else:
183                # CNOT/CZ/SWAP
184                return self._apply_fn(psi, n, d, qudits, dagger)
185
186        # Optimized 1-qudit system case
187        if n == 1:
188            U = self.matrix(d, dagger)
189            return U @ psi
190
191        # General Tensor Contraction
192        Umat = self.dagger_matrix_fn(d) if dagger else self.matrix_fn(d)
193        k = self.n_qudits
194
195        U = Umat.reshape((d,) * k + (d,) * k)
196        psi_nd = psi.reshape((d,) * n)
197        axes_U = list(range(k, 2 * k))
198        axes_psi = list(qudits)
199        contracted = np.tensordot(U, psi_nd, axes=(axes_U, axes_psi))
200        rest = [i for i in range(n) if i not in qudits]
201
202        perm = []
203        for i in range(n):
204            if i in qudits:
205                perm.append(qudits.index(i))
206            else:
207                perm.append(k + rest.index(i))
208
209        out = np.transpose(contracted, perm)
210        return out.reshape(-1)

330        Apply the noise channel to a density matrix using using the
331        Kraus operator sum representation. Implements on-the-fly computation
332        and caching of Kraus operators.
333
334        rho' = sum(E_k * rho * E_k†)
335
336        Parameters
337        ----------
338        rho : np.ndarray
339            Density matrix of shape (d^n, d^n).
340        circuit : QuantumCircuit

344        dagger : bool
345            Ignored for noise channels.
346
347        Returns
348        -------
349        np.ndarray
350            Updated density matrix.
351        """
352        # Check the circuit's cache for our Kraus operators.
353        kraus_operators = circuit._kraus_operator_cache.get(self.model_id)
354
355        # If not in the cache, compute and store them.
356        if kraus_operators is None:
357            # Retrieve the probability array from the permanent registry.
358            prob_array = circuit.noise_registry.get(self.model_id)
359            if prob_array is None:
360                # This should not happen if the circuit is constructed correctly.
361                return rho
362
363            d = circuit.d
364            prob_matrix = prob_array.reshape((d, d))
365
366            # Compute the list of d x d Kraus operators.
367            kraus_operators = []
368            for a in range(d):
369                for b in range(d):
370                    prob = prob_matrix[a, b]
371                    if (
372                        prob > 1e-12
373                    ):  # Only create operators for non-zero probabilities.
374                        # E_k = sqrt(p_k) * W_k
375                        kraus_op = np.sqrt(prob) * gates.W(d, a, b)
376                        kraus_operators.append(kraus_op)
377
378            # Store the computed list in the cache for future use in this run.
379            circuit._kraus_operator_cache[self.model_id] = kraus_operators
380
381        # Apply the Kraus sum: rho' = sum(E_k * rho * E_k_dagger)
382        rho_new = np.zeros_like(rho)
383        for E_k in kraus_operators:
384            term = Gate._apply_operator_to_density_matrix(
385                rho, E_k, qudits, circuit.n, circuit.d
386            )
387            rho_new += term
388
389        return rho_new
